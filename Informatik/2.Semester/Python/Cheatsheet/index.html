3.1 Person · BesserePerson · __repr__
class Person:
    def __init__(self, name, height):
        self.name = name
        self.height = height

    def grow(self, amount):
        self.height += amount

    def __repr__(self):                  # muss str RETURNEN
        cls = self.__class__.__name__    # Extra: Klassenname
        return f"{cls}({self.name}, {self.height})"

class BesserePerson(Person):             # erbt __init__/__repr__
    def grow(self, amount):
        if not 0 < amount < 10:
            raise ValueError(f"0 < amount < 10, war {amount}")
        super().grow(amount)             # Eltern-Methode

persons = {n: Person(n, h) for n, h in zip(names, heights)}
print(persons["Anna"])             # Person(Anna, 1.68)
print(BesserePerson("Max", 180))   # BesserePerson(Max, 180)
p = BesserePerson("Max", 180)
try:
    p.grow(12)
except ValueError as e:
    print("Fehler:", e)
